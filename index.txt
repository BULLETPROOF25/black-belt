/* =====================================================
   SIMPLE LOCAL PASSWORD HASH

   This avoids the previous login problem with
   crypto.subtle when opening an HTML file directly.

   NOTE:
   This is for a local learning app, NOT a production
   authentication server.
===================================================== */

function hashPassword(password){

    let hash = 2166136261;

    for(let i=0;i<password.length;i++){

        hash ^= password.charCodeAt(i);

        hash +=
            (hash << 1) +
            (hash << 4) +
            (hash << 7) +
            (hash << 8) +
            (hash << 24);

    }

    return (
        hash >>> 0
    ).toString(16);
}

function setupAdmin(){

    let admin;

    try{

        admin=JSON.parse(
            localStorage.getItem(ADMIN_KEY)
        );

    }catch{

        admin=null;

    }

    if(!admin){

        admin={
            username:"admin",

            passwordHash:
                hashPassword(
                    "change-this-admin-password"
                )
        };

        localStorage.setItem(
            ADMIN_KEY,
            JSON.stringify(admin)
        );
    }
}

function isAdmin(){

    const current=getCurrentUser();

    let admin;

    try{

        admin=JSON.parse(
            localStorage.getItem(ADMIN_KEY)
        );

    }catch{

        return false;

    }

    return(
        admin &&
        current===admin.username
    );
}

function createAccount(){

        document.getElementById("username")

        document.getElementById("password")
        .value;

    if(username.length<3){

        showMessage(
            "Username must be at least 3 characters."
        );

        return;
    }

    if(username.toLowerCase()==="admin"){

        showMessage(
            "That username is reserved for the creator."
        );

        return;
    }

    if(password.length<8){

        showMessage(
            "Password must be at least 8 characters."
        );

        return;
    }

    const users=getUsers();

    if(users[username]){

        showMessage(
            "That username is already taken. Choose another."
        );

        return;
    }

    users[username]={

        passwordHash:hashPassword(password),

        createdAt:new Date().toISOString()

    };

    saveUsers(users);

    localStorage.setItem(
        CURRENT_USER_KEY,
        username
    );

    showMessage("");

    startApp();
}


/* =====================================================
   LOGIN
===================================================== */

function login(){

    const username=
        document.getElementById("username")
        .value
        .trim();

    const password=
        document.getElementById("password")
        .value;

    if(!username || !password){

        showMessage(
            "Enter your username and password."
        );

        return;
    }


    /* Creator login */

    let admin;

    try{

        admin=JSON.parse(
            localStorage.getItem(ADMIN_KEY)
        );

    }catch{

        admin=null;

    }


    if(
        admin &&
        username===admin.username &&
        hashPassword(password)===admin.passwordHash
    ){

        localStorage.setItem(
            CURRENT_USER_KEY,
            username
        );

        startApp();

        return;
    }


    /* Normal user login */

    const users=getUsers();

    if(!users[username]){

        showMessage(
            "Username or password is incorrect."
        );

        return;
    }

    if(
        hashPassword(password)!==
        users[username].passwordHash
    ){

        showMessage(
            "Username or password is incorrect."
        );

        return;
    }


    localStorage.setItem(
        CURRENT_USER_KEY,
        username
    );

    startApp();
}

/* =====================================================
   COURSE PAGE
===================================================== */

function showCourse(courseKey){

    const course=courses[courseKey];

    const progress=
        getProgress(getCurrentUser());

    let html=`

        <button onclick="home()">
            ? Back
        </button>

        <h1 class="pageTitle">
            ${course.name}
        </h1>

        <p>
            ${course.description}
        </p>

    `;


    course.levels.forEach(
        (level,levelIndex)=>{

            const unlocked=
                levelIndex===0 ||
                previousLevelComplete(
                    courseKey,
                    levelIndex,
                    progress
                );


            html+=`

                <div
                    class="card"
                    style="margin-top:25px"
                >

                    <h2>

                        ${unlocked ? "??" : "??"}

                        ${level.name}

                    </h2>

                    <div
                        class="belt ${level.belt}">
                        ${level.name}
                    </div>

            `;


            level.lessons.forEach(
                (lesson,lessonIndex)=>{

                    const id=
                        lessonId(
                            courseKey,
                            levelIndex,
                            lessonIndex
                        );

                    const completed=
                        progress[id]===true;


                    html+=`

                        <div
                            class="lesson
                            ${!unlocked ? "locked":""}
                            ${completed ? "complete":""}"
                        >

                            <div class="lessonTitle">

                                ${
                                    completed
                                    ? "? "
                                    : ""
                                }

                                ${lesson.title}

                            </div>


                            <p>

                                ${
                                    completed
                                    ? "Lesson completed."
                                    : unlocked
                                    ? "Watch the assigned lesson, then complete it."
                                    : "Complete the previous level first."
                                }

                            </p>


                            <button
                                class="${
                                    unlocked
                                    ? "videoBtn"
                                    : "disabledBtn"
                                }"

                                ${
                                    unlocked
                                    ? `onclick="openVideo('${lesson.video}')"`
                                    : "disabled"
                                }
                            >

                                ?? Watch Assigned Video

                            </button>


                            <button
                                class="${
                                    unlocked && !completed
                                    ? "completeBtn"
                                    : "disabledBtn"
                                }"

                                ${
                                    unlocked && !completed
                                    ? `onclick="completeLesson('${courseKey}',${levelIndex},${lessonIndex})"`
                                    : "disabled"
                                }
                            >

                                ${
                                    completed
                                    ? "Completed ?"
                                    : "Complete Lesson"
                                }

                            </button>

                        </div>

                    `;
                }
            );


            html+=`</div>`;
        }
    );


    document.getElementById("content")
        .innerHTML=html;
}


/* =====================================================
   LEVEL SYSTEM
===================================================== */

function lessonId(
    course,
    level,
    lesson
){

    return course+"_"+level+"_"+lesson;
}


function previousLevelComplete(
    courseKey,
    levelIndex,
    progress
){

    if(levelIndex===0){
        return true;
    }

    const previousLevel=
        courses[courseKey]
        .levels[levelIndex-1];


    return previousLevel.lessons.every(
        (_,lessonIndex)=>{

            return progress[
                lessonId(
                    courseKey,
                    levelIndex-1,
                    lessonIndex
                )
            ]===true;

        }
    );
}


/* =====================================================
   COMPLETE LESSON
===================================================== */

function completeLesson(
    courseKey,
    levelIndex,
    lessonIndex
){

    const username=
        getCurrentUser();

    const progress=
        getProgress(username);


    progress[
        lessonId(
            courseKey,
            levelIndex,
            lessonIndex
        )
    ]=true;


    saveProgress(
        username,
        progress
    );


    showCourse(courseKey);
}

            </div>

            <p>
                ${completed}

                ${total}
                lessons completed.
            </p>

        </div>

        <div
            class="grid"

    for(const key in courses){

        let courseTotal=0;

                level.lessons.forEach(

                                    levelIndex,

                    }

/* =====================================================
   HTML SAFETY
===================================================== */

function escapeHTML(value){

    return String(value)
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");
}
